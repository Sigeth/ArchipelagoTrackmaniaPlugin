#Include "TextLib" as TL
#Include "Modes/TrackMania/Libs/Archipelago/EffectsStateManager.Script.txt" as AP_EffSM

#Const C_AP_EventType "AP.Effect"

Void ApplyEffect(Text EffectId, Text[] Args) {
	foreach (_Player in AllPlayers) {
		log("[Archipelago] Playing effect " ^ EffectId ^ " on " ^ _Player.User.Login);
		switch(EffectId) {
			case "test": {
				log("[Archipelago] test effect played");
				declare FakeVariable = 3;
			}
			case "tinycar": {
				// TODO: works weirdly, not finished yet
				_Player.TinyCar = True;
				log("[Archipelago] tinycar effect played");
			}
			case AP_EffSM::C_AP_Trap_LowGravity: {
				_Player.GravityCoef = 0.5;
				AP_EffSM::ArmEffect(_Player, EffectId, 30000);
				log("[Archipelago] lowgravity effect played");
			}
		}
	}
}

Void DispatchEffect() {
	foreach (Event in UIManager.PendingEvents) {
		if (Event.Type == CUIConfigEvent::EType::OnLayerCustomEvent) {
			if (Event.CustomEventType == C_AP_EventType) {
				declare Text EffectId = Event.CustomEventData[0];
				declare Text[] Args;
				for (i, 1, Event.CustomEventData.count - 1) {
					Args.add(Event.CustomEventData[i]);
				}
				ApplyEffect(EffectId, Args);
			}
		}
	}
}
